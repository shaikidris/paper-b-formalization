'PaperB.Digraph.siblingBlockSystem_root_path_avoiding' depends on axioms: [propext, Classical.choice, Quot.sound]
@PaperB.Digraph.siblingBlockSystem_root_path_avoiding : ∀ {g : PaperB.Digraph} {r d : Fin g.vertexCount}
  {rank : Fin g.vertexCount → Nat} {parent : Fin g.vertexCount → Fin g.vertexCount}
  (progress : ∀ (x y : Fin g.vertexCount), g.HasArc x y → rank x < rank y)
  (injective : ∀ ⦃x y : Fin g.vertexCount⦄, rank x = rank y → x = y) (spec : g.ParentSpec r parent)
  (hd : g.InitiallyReachable r d) (b t : g.SiblingBlockKey r d parent),
  b ≠ PaperB.Digraph.SiblingBlockKey.root g r d parent →
    t ≠ b →
      have S := PaperB.Digraph.siblingBlockSystem progress injective spec hd;
      PaperB.AvoidPath S.QuotientEdge (fun x => x = b) (PaperB.Digraph.SiblingBlockKey.root g r d parent) t
