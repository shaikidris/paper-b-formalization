@buildProperLossForest_ancestor_iff_interval_of_node : ∀ {H : LabelledQuotient} {rank : Fin H.vertexCount → Nat},
  (∀ (u v : Fin H.vertexCount), H.HasQuotientArc u v → rank u < rank v) →
    H.IsFlat →
      ∀ {a : Fin ((H.buildProperLossForest rank).realNodeCount + 1)} {P : H.CutPair},
        (H.buildProperLossForest rank).nodeKey? a = some P →
          ∀ (y : Fin H.vertexCount),
            let F := H.buildProperLossForest rank;
            IsForestAncestor F.parent a (F.label y) ↔ dfsIntervalContains F.rootNode F.parent F.depth a (F.label y)
@buildProperLossForest_properLoss_iff_interval_of_node : ∀ {H : LabelledQuotient} {rank : Fin H.vertexCount → Nat},
  (∀ (u v : Fin H.vertexCount), H.HasQuotientArc u v → rank u < rank v) →
    (∀ (a b : Fin H.vertexCount), rank a = rank b → a = b) →
      H.IsFlat →
        ∀ {a : Fin ((H.buildProperLossForest rank).realNodeCount + 1)} {P : H.CutPair},
          (H.buildProperLossForest rank).nodeKey? a = some P →
            ∀ (y : Fin H.vertexCount),
              let F := H.buildProperLossForest rank;
              H.ProperLoss P y ↔ dfsIntervalContains F.rootNode F.parent F.depth a (F.label y)
