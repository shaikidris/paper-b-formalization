def PaperB.OracleContract.Candidate : PaperB.OracleContract.CandidateInterface → Nat → Nat → Prop :=
fun M buildExp queryExp =>
  ∃ build query C,
    0 < C ∧
      ∀ (g : PaperB.Digraph) (r : Fin g.vertexCount),
        g.IsAcyclic →
          ∃ b,
            M.observeBuild build g r = some b ∧
              b.operations ≤ C * ((g.vertexCount + g.arcCount) * PaperB.OracleContract.L g.vertexCount ^ buildExp) ∧
                b.peakLiveWords ≤ C * (g.vertexCount + g.arcCount) ∧
                  b.words.size ≤ b.retainedWords ∧
                    b.retainedWords ≤ C * g.vertexCount ∧
                      ∀ (F : PaperB.FailureSet g.vertexCount) (t : Fin g.vertexCount),
                        ∃ q,
                          M.observeQuery query (b.index g.vertexCount) F t = some q ∧
                            q.operations ≤ C * PaperB.OracleContract.L g.vertexCount ^ queryExp ∧
                              (q.answer = true ↔ g.ReachableAvoiding F r t)
